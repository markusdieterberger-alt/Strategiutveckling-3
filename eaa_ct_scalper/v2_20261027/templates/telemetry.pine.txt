// Diagnostic-only events. Never count these as fills, trades, or funded PnL.
// Mask bits: long FADE=1,DIV=2,TRAP=4,AGG=8,ABSORB=16,VRZ=32; short=long*64.
// Raw=winning candidate after original model priority and TP-distance gate.
// Eval/flow/skipped reveal reachability, not counterfactual skipped signals.
// Block reason: 0 no raw,1 unconfirmed,2 dates,3 conflict,4 busy,5 time,6 FP,
// 7 invalid levels,8 submitted. Force-flat has precedence over entry dispatch.
int v2BlockReason = not eaaRawAny ? 0 : not barstate.isconfirmed ? 1 : not eaaInDates ? 2 : eaaConflict ? 3 : not eaaSlotFree ? 4 : not eaaTimeOk ? 5 : not eaaDataOk ? 6 : not v2LevelsValid ? 7 : eaaOrderSubmitted ? 8 : 9
bool v2TraceNow = barstate.isconfirmed and eaaInDates and (v2TraceAllBars or eaaRawAny or eaaOrderSubmitted or v2ObserveEntry or eaaClosedNow or eaaUnfilledNow)
if v2TraceNow and (v2PineLogs or v2DiagAlerts)
    string diagnostic = '{"event":"DIAGNOSTIC_NOT_A_TRADE","version":"' + v2Version + '","bar_ms":' + str.tostring(time) + ',"close_ms":' + str.tostring(time_close) + ',"realtime":' + (barstate.isrealtime ? "true" : "false") + ',"calc_in_bar":' + str.tostring(v2CalcInBar) + ',"eval_mask":' + str.tostring(v2EvalMask) + ',"flow_mask":' + str.tostring(v2FlowMask) + ',"raw_mask":' + str.tostring(v2RawMask) + ',"priority_skipped":' + str.tostring(v2PrioritySkipped) + ',"filter_mask":' + str.tostring(v2FilterMask) + ',"exec_gates":' + str.tostring(v2ExecGates) + ',"block_reason":' + str.tostring(v2BlockReason) + ',"fp_present":' + (not na(fp) ? "true" : "false") + ',"fp_streak":' + str.tostring(eaaFootprintStreak) + ',"fp_buy":' + f_v2Num(fpBuyVol) + ',"fp_sell":' + f_v2Num(fpSellVol) + ',"fp_delta":' + f_v2Num(fpDelta) + ',"poc":' + f_v2Num(fpPocMid) + ',"poc_fresh":' + (v2PocFresh ? "true" : "false") + ',"vah":' + f_v2Num(fpVAH) + ',"val":' + f_v2Num(fpVAL) + ',"fp_slope":' + f_v2Num(fpDeltaSlope) + ',"htf":' + v2HtfSnapshot + ',"candidate_sl":' + f_v2Num(v2CandidateSl) + ',"candidate_tp":' + f_v2Num(v2CandidateTp) + ',"submitted":' + (eaaOrderSubmitted ?     if v2PineLogs
        log.info(diagnostic)
    if v2DiagAlerts and barstate.isrealtime
        alert(diagnostic, alert.freq_once_per_bar_close)
